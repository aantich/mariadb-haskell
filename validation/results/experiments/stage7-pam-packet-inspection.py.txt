with socket.create_connection((os.environ['MB_AUTH_HOST'],int(os.environ['MB_AUTH_PORT']))) as s:
 def exact(n):
  b=b''
  while len(b)<n:
   chunk=s.recv(n-len(b))
   if not chunk:raise EOFError
   b+=chunk
  return b
 def read():
  h=exact(4);return h[3],exact(int.from_bytes(h[:3],'little'))
