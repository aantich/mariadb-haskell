{-# LANGUAGE OverloadedStrings #-}
import Control.Monad
import Database.MySQL.Base
import qualified Data.Map.Strict as M
import Validation.Common
main = do
  withConn $ \c -> do
    exec c "SET collation_connection='utf8mb4_bin'"
    s <- getSessionState c
    print ("collation",M.lookup "collation_connection" (sessionVariables s))
    rows c "SELECT @@collation_connection" >>= print
  withConn $ \c -> do
    exec c "SET session_track_transaction_info='CHARACTERISTICS'"
    exec c "START TRANSACTION READ ONLY"
    s <- getSessionState c
    print ("initial tx",sessionTransactionCharacteristics s, sessionTransactionState s)
    exec c "SET session_track_transaction_info='STATE'"
    s2 <- getSessionState c
    print ("after downgrade",sessionTransactionCharacteristics s2, sessionTransactionState s2,sessionLastChanges s2)
    exec c "ROLLBACK"
    exec c "START TRANSACTION READ WRITE"
    s3 <- getSessionState c
    print ("new transaction",sessionTransactionCharacteristics s3,sessionTransactionState s3)
    rows c "SELECT @@in_transaction,@@tx_read_only" >>= print
    exec c "ROLLBACK"
