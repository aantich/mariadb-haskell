{-# LANGUAGE OverloadedStrings #-}
module Main where
import Control.Exception
import Control.Monad
import Database.MySQL.Base
import GHC.Clock
import qualified System.IO.Streams as S
import Validation.Common (config, withConnect)

main :: IO ()
main = do
  ci <- config
  forM_ [1024,1,128,1,1024,128] $ \count ->
    withConnect (connectWithOptions defaultConnectionOptions {enableMetadataCaching=False} ci) $ \conn ->
      bracket (replicateM count (prepareStatement conn "SELECT 7")) (mapM_ closePrepared) $ \statements -> do
        let statement=head statements
            operation=queryPrepared statement [] >>= S.skipToEof . snd
        replicateM_ 100 operation
        start <- getMonotonicTimeNSec
        replicateM_ 2000 operation
        end <- getMonotonicTimeNSec
        print ("uncached query ns",count,(end-start) `div` 2000)
