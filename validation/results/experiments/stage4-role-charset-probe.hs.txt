{-# LANGUAGE OverloadedStrings #-}
import Control.Monad
import Data.String
import Database.MySQL.Base
import qualified Database.MySQL.Connection as C
import Data.IORef
import System.Environment
import Validation.Common
main = do
 ci <- config
 user <- getEnv "MB_ROLE_CHARSET_USER"
 other <- getEnv "MB_ROLE_CHARSET_OTHER"
 withConnect (connect ci {ciUser=fromString user,ciCharset=8}) $ \c -> do
  i <- readIORef (C.connectionInitialSession c)
  print (C.initialRole =<< i)
  print =<< rows c "SELECT CURRENT_ROLE(),HEX(CAST(CURRENT_ROLE() AS BINARY))"
  exec c "SET NAMES utf8mb4"
  exec c ("SET ROLE `" ++ other ++ "`")
  exec c "SET NAMES latin1"
  void (resetConnection c)
  print =<< rows c "SELECT CURRENT_ROLE(),HEX(CAST(CURRENT_ROLE() AS BINARY))"
