{-# LANGUAGE OverloadedStrings #-}
{-# LANGUAGE ScopedTypeVariables #-}
import Control.Exception
import Data.Bits
import Data.IORef
import qualified Data.ByteString as B
import qualified Data.ByteString.Lazy as L
import qualified Database.MySQL.Connection as C
import Database.MySQL.Authentication
import Database.MySQL.Protocol.Auth
import Database.MySQL.Protocol.Packet
import qualified System.IO.Streams as S

packet n body = Packet (fromIntegral (B.length body)) n (L.fromStrict body)
hello = Greeting 10 "12.3.3-MariaDB" 1 "12345678" clientCap 224 2 "901234567890" "mysql_native_password"
ok = packet 4 "\0\0\0\2\0\0\0"
main = do
  called <- newIORef False
  replies <- newIORef (0::Int)
  input <- S.fromList [ok]
  conn <- C.newConnection C.defaultConnectionOptions input (\_ -> modifyIORef' replies (+1)) (pure ())
  C.setExchangeSequence conn 3
  let auth=defaultAuthenticationOptions {authenticationDialog=Just (\_ -> writeIORef called True >> pure "secret")}
      oversized=packet 2 ("\254mysql_clear_password\0" <> B.replicate (2*1024*1024) 65)
  result <- try (C.completeAuthWithOptions auth conn hello True "unused" oversized)
  callbackCalled <- readIORef called
  sent <- readIORef replies
  putStrLn ("oversized clear-password challenge: " ++ either (displayException :: SomeException -> String) (const "ACCEPTED") result ++ "; callback=" ++ show callbackCalled ++ "; replies=" ++ show sent)
  bodyRead <- newIORef False
  phase <- newIORef (0::Int)
  let len = 2*1024*1024 :: Int
      header = B.pack [fromIntegral len,fromIntegral (len `shiftR` 8),fromIntegral (len `shiftR` 16),4]
  bytes <- S.makeInputStream $ do
      n <- atomicModifyIORef' phase (\v -> (v+1,v))
      if n == 0 then pure (Just header) else writeIORef bodyRead True >> ioError (userError "BODY_READ_REQUESTED")
  conn2 <- C.newByteConnection C.defaultConnectionOptions bytes (\_ -> pure ()) (pure ())
  C.setExchangeSequence conn2 3
  let first=packet 2 "\254dialog\0\4Password:"
  second <- try (C.completeAuthWithOptions auth conn2 hello True "unused" first)
  attempted <- readIORef bodyRead
  putStrLn ("oversized continuation header: " ++ either (displayException :: SomeException -> String) (const "ACCEPTED") second ++ "; attemptedBodyRead=" ++ show attempted)
