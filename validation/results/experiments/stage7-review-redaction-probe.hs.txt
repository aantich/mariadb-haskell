{-# LANGUAGE OverloadedStrings #-}
{-# LANGUAGE ScopedTypeVariables #-}
import Control.Exception
import Data.IORef
import Data.List (isInfixOf)
import qualified Data.ByteString as B
import qualified Data.ByteString.Lazy as L
import qualified Database.MySQL.Connection as C
import Database.MySQL.Authentication
import Database.MySQL.Protocol.Auth
import Database.MySQL.Protocol.Packet
import qualified System.IO.Streams as S
packet n body = Packet (fromIntegral (B.length body)) n (L.fromStrict body)
hello = Greeting 10 "12.3.3-MariaDB" 1 "12345678" clientCap 224 2 "901234567890" "mysql_native_password"
probe label firstBody response = do
  input <- S.fromList [packet 4 response]
  conn <- C.newConnection C.defaultConnectionOptions input (const (pure ())) (pure ())
  C.setExchangeSequence conn 3
  let auth=defaultAuthenticationOptions {authenticationDialog=Just (passwordDialog "secret")}
  result <- try (C.completeAuthWithOptions auth conn hello True "unused" (packet 2 firstBody))
  case result of
    Left (err :: SomeException) -> putStrLn (label ++ ": markerDisclosed=" ++ show ("PRIVATE-TOKEN-MARKER" `isInfixOf` displayException err) ++ "; " ++ displayException err)
    Right () -> putStrLn (label ++ ": unexpectedly succeeded")
